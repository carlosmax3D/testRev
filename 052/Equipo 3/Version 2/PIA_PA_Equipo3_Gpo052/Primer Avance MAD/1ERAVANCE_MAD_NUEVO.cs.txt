    public class FormBase : Form
    {
        protected FormBase(string titulo, int ancho, int alto)
        {
            Text = titulo;
            ClientSize = new Size(ancho, alto);  
            StartPosition = FormStartPosition.CenterScreen;
            FormBorderStyle = FormBorderStyle.FixedDialog;
            MaximizeBox = false;
            MinimizeBox = false;
        }

        protected Label Etiqueta(string texto, int x, int y, int ancho)
        {
            Label l = new Label { Text = texto, Location = new Point(x, y + 3), Width = ancho, AutoSize = false };
            Controls.Add(l);
            return l;
        }

        protected TextBox Texto(int x, int y, int ancho)
        {
            TextBox t = new TextBox { Location = new Point(x, y), Width = ancho };
            Controls.Add(t);
            return t;
        }

        protected ComboBox Combo(int x, int y, int ancho)
        {
            ComboBox c = new ComboBox { Location = new Point(x, y), Width = ancho, DropDownStyle = ComboBoxStyle.DropDownList };
            Controls.Add(c);
            return c;
        }

        protected DateTimePicker Fecha(int x, int y, int ancho)
        {
            DateTimePicker d = new DateTimePicker { Location = new Point(x, y), Width = ancho, Format = DateTimePickerFormat.Short };
            Controls.Add(d);
            return d;
        }

        // Etiqueta + caja de texto / combo / fecha (la caja queda 110 px a la derecha de la etiqueta)
        protected TextBox CampoTexto(string etiqueta, int x, int y)
        {
            Etiqueta(etiqueta, x, y, 105);
            return Texto(x + 110, y, 200);
        }

        protected ComboBox CampoCombo(string etiqueta, int x, int y)
        {
            Etiqueta(etiqueta, x, y, 105);
            return Combo(x + 110, y, 200);
        }

        protected DateTimePicker CampoFecha(string etiqueta, int x, int y)
        {
            Etiqueta(etiqueta, x, y, 105);
            return Fecha(x + 110, y, 200);
        }

        protected Button Boton(string texto, int x, int y, int ancho)
        {
            Button b = new Button { Text = texto, Location = new Point(x, y), Width = ancho };
            Controls.Add(b);
            return b;
        }

        protected Label Recuadro(string texto, int x, int y, int ancho, int alto)
        {
            Label l = new Label
            {
                Text = texto,
                Location = new Point(x, y),
                Size = new Size(ancho, alto),
                BorderStyle = BorderStyle.FixedSingle,
                AutoSize = false
            };
            Controls.Add(l);
            return l;
        }

        protected DataGridView Tabla(int x, int y, int ancho, int alto, params string[] columnas)
        {
            DataGridView t = new DataGridView
            {
                Location = new Point(x, y),
                Size = new Size(ancho, alto),
                AllowUserToAddRows = false,
                ReadOnly = true,
                MultiSelect = false,
                RowHeadersVisible = false,
                SelectionMode = DataGridViewSelectionMode.FullRowSelect,
                AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill
            };
            for (int i = 0; i < columnas.Length; i++)
                t.Columns.Add("col" + i, columnas[i]);
            Controls.Add(t);
            return t;
        }
    }

    // 1. frmLogin
    public class frmLogin : FormBase
    {
        private Label lblCorreo;
        private Label lblContrasena;
        private TextBox txtCorreo;
        private TextBox txtContrasena;
        private Button btnIngresar;
        private Button btnSalir;

        public frmLogin() : base("Login", 350, 180)
        {
            InitializeComponent();
        }

        private void InitializeComponent()
        {
            lblCorreo = Etiqueta("Correo", 20, 20, 100);
            txtCorreo = Texto(130, 20, 180);

            lblContrasena = Etiqueta("Contraseña", 20, 60, 100);
            txtContrasena = Texto(130, 60, 180);
            txtContrasena.UseSystemPasswordChar = true;

            btnIngresar = Boton("Ingresar", 60, 110, 90);
            btnSalir = Boton("Salir", 180, 110, 90);

            AcceptButton = btnIngresar;   // Enter = Ingresar
            CancelButton = btnSalir;      // Esc = Salir

            btnIngresar.Click += btnIngresar_Click;
            btnSalir.Click += (s, e) => Close();
        }

        private void btnIngresar_Click(object sender, EventArgs e)
        {
            if (txtCorreo.Text.Trim() == "" || txtContrasena.Text == "")
            {
                MessageBox.Show("Captura tu correo y contraseña.", "Login",
                    MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }

            // TODO: validar usuario contra la base de datos.
            Hide();
            using (frmMenuPrincipal menu = new frmMenuPrincipal())
            {
                menu.ShowDialog();
            }
            Close();
        }
    }

    // 2. frmMenuPrincipal
    public class frmMenuPrincipal : Form
    {
        private MenuStrip menuStrip;
        private Label lblBienvenida;

        public frmMenuPrincipal()
        {
            InitializeComponent();
        }

        private void InitializeComponent()
        {
            Text = "Menú Principal";
            ClientSize = new Size(800, 460);
            StartPosition = FormStartPosition.CenterScreen;

            lblBienvenida = new Label
            {
                Text = "Sistema de Venta de Boletos",
                Dock = DockStyle.Fill,
                TextAlign = ContentAlignment.MiddleCenter,
                Font = new Font("Segoe UI", 18F, FontStyle.Bold)
            };
            Controls.Add(lblBienvenida);   // se agrega primero para que el menú quede arriba

            menuStrip = new MenuStrip();

            // Opciones del menú exactamente como en el documento (11)
            menuStrip.Items.Add(CrearOpcion("Inicio"));
            menuStrip.Items.Add(CrearOpcion("Usuarios"));
            menuStrip.Items.Add(CrearOpcion("Clientes"));
            menuStrip.Items.Add(CrearOpcion("Estadios"));

            // Zonas: desplegable para poder llegar también a Asientos
            ToolStripMenuItem zonas = new ToolStripMenuItem("Zonas");
            zonas.DropDownItems.Add(CrearOpcion("Zonas"));
            zonas.DropDownItems.Add(CrearOpcion("Asientos"));
            menuStrip.Items.Add(zonas);

            menuStrip.Items.Add(CrearOpcion("Partidos"));
            menuStrip.Items.Add(CrearOpcion("Venta Boletos"));
            menuStrip.Items.Add(CrearOpcion("Cancelaciones"));
            menuStrip.Items.Add(CrearOpcion("Facturación"));

            // Reportes: desplegable para llegar a los 3 reportes
            ToolStripMenuItem reportes = new ToolStripMenuItem("Reportes");
            reportes.DropDownItems.Add(CrearOpcion("Reporte de Ventas"));
            reportes.DropDownItems.Add(CrearOpcion("Reporte de Ocupación"));
            reportes.DropDownItems.Add(CrearOpcion("Historial de Cliente"));
            menuStrip.Items.Add(reportes);

            menuStrip.Items.Add(CrearOpcion("Salir"));

            MainMenuStrip = menuStrip;
            Controls.Add(menuStrip);
        }

        private ToolStripMenuItem CrearOpcion(string texto)
        {
            ToolStripMenuItem item = new ToolStripMenuItem(texto);
            item.Click += AbrirPantalla;
            return item;
        }

        private void AbrirPantalla(object sender, EventArgs e)
        {
            string opcion = ((ToolStripMenuItem)sender).Text;
            Form pantalla = null;

            switch (opcion)
            {
                case "Usuarios": pantalla = new frmUsuarios(); break;
                case "Clientes": pantalla = new frmClientes(); break;
                case "Estadios": pantalla = new frmEstadios(); break;
                case "Zonas": pantalla = new frmZonas(); break;
                case "Asientos": pantalla = new frmAsientos(); break;
                case "Partidos": pantalla = new frmPartidos(); break;
                case "Venta Boletos": pantalla = new frmVentaBoletos(); break;
                case "Cancelaciones": pantalla = new frmCancelaciones(); break;
                case "Facturación": pantalla = new frmFacturacion(); break;
                case "Reporte de Ventas": pantalla = new frmReporteVentas(); break;
                case "Reporte de Ocupación": pantalla = new frmReporteOcupacion(); break;
                case "Historial de Cliente": pantalla = new frmHistorialCliente(); break;
                case "Salir": Close(); return;
                default: return;   // "Inicio" y "Reportes" no abren pantalla
            }

            pantalla.ShowDialog(this);
            pantalla.Dispose();
        }
    }
